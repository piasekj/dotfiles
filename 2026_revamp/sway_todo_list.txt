Yazi intall directly from github:
sudo dnf install curl unzip file

mkdir -p ~/.local/bin

curl -fL https://github.com/sxyazi/yazi/releases/latest/download/yazi-x86_64-unknown-linux-gnu.zip -o /tmp/yazi.zip

unzip -o /tmp/yazi.zip -d /tmp

cp /tmp/yazi-x86_64-unknown-linux-gnu/{yazi,ya} ~/.local/bin/

chmod +x ~/.local/bin/{yazi,ya}
